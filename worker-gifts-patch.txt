function giftConfigForProduct(product) {
  const g = product?.giftConfig;
  if (!g || g.enabled !== true || !Array.isArray(g.gifts) || !g.gifts.length) return null;
  return {
    enabled: true,
    mode: g.mode === 'choose' ? 'choose' : 'forced',
    maxChoices: Math.min(10, Math.max(1, Number(g.maxChoices) || 1)),
    giftBudgetWholesale: Math.max(0, Number(g.giftBudgetWholesale) || 0),
    gifts: [...new Set(g.gifts.map(String))]
  };
}

if (url.pathname === '/api/gifts/quote' && request.method === 'POST') {
  try {
    const body = await request.json();
    const products = await getProducts(env);
    const parent = products.find(p => String(p.id) === String(body.productId));
    if (!parent) return json({error:'المنتج غير موجود.'},404);

    const cfg = giftConfigForProduct(parent);
    if (!cfg) return json({error:'لا توجد هدايا مفعلة لهذا المنتج.'},400);

    const quantity = Math.max(1, Number(body.quantity) || 1);
    let ids = Array.isArray(body.giftIds) ? body.giftIds.map(String) : [];
    const allowed = new Set(cfg.gifts.filter(id => id !== String(parent.id)));

    if (cfg.mode === 'forced') ids = cfg.gifts.slice(0, cfg.maxChoices);
    ids = [...new Set(ids)].filter(id => allowed.has(id)).slice(0, cfg.maxChoices);

    if (cfg.mode === 'choose' && !ids.length) {
      return json({error:'اختار هدية واحدة على الأقل.'},400);
    }

    const selected = ids.map(id => products.find(p => String(p.id) === id)).filter(Boolean);
    const values = {};
    let unitGiftValue = 0;
    for (const p of selected) {
      const wholesale = Math.max(0, Number(p.wholesalePrice) || 0);
      values[String(p.id)] = wholesale;
      unitGiftValue += wholesale;
    }

    if (cfg.giftBudgetWholesale > 0 && unitGiftValue > cfg.giftBudgetWholesale) {
      return json({
        error:`إجمالي الهدايا ${unitGiftValue} ج بسعر الجملة، بينما الحد المسموح ${cfg.giftBudgetWholesale} ج.`
      },400);
    }

    return json({
      success:true,
      productId:parent.id,
      giftIds:selected.map(p => p.id),
      values,
      unitGiftValue,
      giftValue:unitGiftValue * quantity,
      budget:cfg.giftBudgetWholesale,
      maxChoices:cfg.maxChoices
    });
  } catch (error) {
    return json({error:String(error?.message || error || 'تعذر التحقق من الهدايا.')},500);
  }
}
